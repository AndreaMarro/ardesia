   181	async function probeExperiment(page, expId, consoleErrors, pageErrors) {
   182	  // Mount via __ELAB_API
   183	  const mountResult = await page.evaluate(async (id) => {
   184	    const api = window.__ELAB_API;
   185	    if (!api) return { apiReady: false, mounted: false, mountError: 'no __ELAB_API on window' };
   186	    if (typeof api.mountExperiment !== 'function') {
   187	      return { apiReady: true, mounted: false, mountError: 'mountExperiment not a function' };
   188	    }
   189	    let mounted = false;
   190	    let mountError = null;
   191	    try {
   192	      const result = await api.mountExperiment(id);
   193	      mounted = result !== false; // some versions return false on "not found"
   194	    } catch (e) {
   195	      mountError = String(e).slice(0, 300);
   196	    }
   197	    return { apiReady: true, mounted, mountError };
   198	  }, expId);
   199	
   200	  // Settle after mount
   201	  await page.waitForTimeout(SETTLE_MS);
   202	
   203	  // DOM + circuit diagnostics
   204	  const diagnostics = await page.evaluate(() => {
   205	    const api = window.__ELAB_API;
   206	    let circuitState = null;
   207	    try { circuitState = api ? api.getCircuitState() : null; } catch (_) {}
   208	
   209	    const componentNodes = document.querySelectorAll('[data-component-id]');
   210	    const svgEls = document.querySelectorAll('svg');
   211	    const canvasEl = document.querySelector(
   212	      '[data-testid="simulator-canvas"], .simulator-canvas, #simulator-canvas, [class*="SimulatorCanvas"]'
   213	    );
   214	
   215	    // Circuit state diagnostics
   216	    const stateComponents = circuitState?.components
   217	      ? (Array.isArray(circuitState.components)
   218	          ? circuitState.components.length
   219	          : Object.keys(circuitState.components).length)
   220	      : 0;
   221	    const stateConnections = Array.isArray(circuitState?.connections)
   222	      ? circuitState.connections.length
   223	      : (Array.isArray(circuitState?.wires) ? circuitState.wires.length : 0);
   224	
   225	    return {

   227	      svgCount: svgEls.length,
   228	      hasSvg: svgEls.length > 0,
   229	      hasCanvas: !!canvasEl,
   230	      circuitComponentCount: stateComponents,
   231	      circuitConnectionCount: stateConnections,

   233	    };
   234	  });

   239	// ── Classify severity ──────────────────────────────────────────────────────
   240	function classifySeverity(probe, exp, consoleErrors, pageErrors) {
   241	  // P0: API not present, or mount failed, or zero components rendered anywhere
   242	  if (!probe.apiReady) return 'P0_BROKEN';
   243	  if (!probe.mounted) return 'P0_BROKEN';
   244	  if (probe.domComponentCount === 0 && probe.circuitComponentCount === 0) return 'P0_BROKEN';
   245	
   246	  // P0: critical page errors (JS crashes)
   247	  if (pageErrors.length > 0) return 'P0_BROKEN';
   248	
   249	  // P1: components render but connections missing when expected
   250	  const actualComponents = Math.max(probe.domComponentCount, probe.circuitComponentCount);
   251	  const compRatio = exp.expectedComponents > 0 ? actualComponents / exp.expectedComponents : 1;
   252	  const connectionsMissing = exp.expectedConnections > 0 && probe.circuitConnectionCount === 0;
   253	
   254	  if (connectionsMissing) return 'P1_PARTIAL';
   255	  if (compRatio < 0.5) return 'P1_PARTIAL';
   256	
   257	  // P2: OK
   258	  return 'P2_OK';
   259	}
