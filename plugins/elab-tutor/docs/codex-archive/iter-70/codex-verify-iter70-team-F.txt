   109	function findIntentTagEnd(text: string, start: number): number {
   110	  const jsonStart = text.indexOf('{', start);
   111	  if (jsonStart === -1) return -1;
   112	  let depth = 0;
   113	  let inString = false;
   114	  let escaped = false;
   115	  for (let i = jsonStart; i < text.length; i++) {
   116	    const ch = text[i];
   117	    if (escaped) {
   118	      escaped = false;
   119	      continue;
   120	    }
   121	    if (ch === '\\') {
   122	      escaped = true;
   123	      continue;
   124	    }
   125	    if (ch === '"') {
   126	      inString = !inString;
   127	      continue;
   128	    }
   129	    if (inString) continue;
   130	    if (ch === '{') depth++;
   131	    if (ch === '}') {
   132	      depth--;
   133	      if (depth === 0) {
   134	        return text.indexOf(']', i);
   135	      }
   136	    }
   137	  }
   138	  return -1;
   139	}

   141	export function splitControlTags(text: string): ControlSplit {
   142	  const source = String(text || '');
   143	  const tags: string[] = [];
   144	  let prose = '';
   145	  let i = 0;
   146	
   147	  while (i < source.length) {
   148	    if (startsWithAt(source, i, '[AZIONE:')) {
   149	      const end = source.indexOf(']', i);
   150	      if (end !== -1) {
   151	        tags.push(source.slice(i, end + 1));
   152	        i = end + 1;
   153	        continue;
   154	      }
   155	    }
   156	    if (startsWithAt(source, i, '[INTENT:{')) {
   157	      const end = findIntentTagEnd(source, i);
   158	      if (end !== -1) {
   159	        tags.push(source.slice(i, end + 1));
   160	        i = end + 1;
   161	        continue;
   162	      }
   163	    }
   164	    prose += source[i];
   165	    i++;
   166	  }
   167	
   168	  return { prose: prose.replace(/\s+/g, ' ').trim(), tags };
   169	}

   171	export function capWordsPreservingControlTags(text: string, maxWords = UNLIM_MAX_VISIBLE_WORDS): string {
   172	  const { prose, tags } = splitControlTags(text);

   177	  return [cappedProse, ...tags].filter(Boolean).join(' ').trim();

   246	    if (text.includes(id)) ids.add(id);
   247	  }
   248	  if (ids.size === 0) {
   249	    const off = (circuitState?.components || []).find(c => /off|spento|error|errore/i.test(String(c.state || '')));
   250	    if (off?.id) ids.add(off.id);
   251	  }
   252	  return Array.from(ids).slice(0, 3).join(',');
   253	}
