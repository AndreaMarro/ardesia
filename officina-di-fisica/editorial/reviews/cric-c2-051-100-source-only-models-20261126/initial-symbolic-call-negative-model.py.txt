def exact(*values):
    if any(type(v)is bool or not isinstance(v,(int,Fraction,sp.Expr))for v in values):raise ValueError('Exact real numeric data required before cast')
    out=tuple(map(sp.sympify,values))
    if any(v.is_number is not True or v.is_real is not True or v.is_finite is not True or v.has(sp.Float)for v in out):raise ValueError('Finite real exact data required')
    return out
def positive(*values):
    out=exact(*values)
    if any(v<=0 for v in out):raise ValueError('Strictly positive data required')
    return out
def nonnegative(*values):
    out=exact(*values)
    if any(v<0 for v in out):raise ValueError('Nonnegative data required')
    return out
def continuity_velocity(flow,area):
    flow,=exact(flow);area,=positive(area);return flow/area
def bernoulli_pressure_change(rho,vi,vf,g=10,dz=0):
    rho,g=positive(rho,g);vi,vf=nonnegative(vi,vf);dz,=exact(dz)
    return rho*(vi**2-vf**2)/2-rho*g*dz
def stagnation_speed(dp,rho):
    dp,=nonnegative(dp);rho,=positive(rho);return sp.sqrt(2*dp/rho)
def speed_from_pressure_drop(vi,rho,drop):
    vi,=nonnegative(vi);rho,=positive(rho);drop,=exact(drop);square=vi**2+2*drop/rho
    if square<0:raise ValueError('Unreachable speed square in the stated ideal model')
    return sp.sqrt(square)
def area_ratio(vi,vf):

def poiseuille_profile(vmax,r,R):
    vmax,r=nonnegative(vmax,r);R,=positive(R)
    if r>R:raise ValueError('Radial point must be inside or on tube wall')
    return vmax*(1-r**2/R**2)
def mean_poiseuille(vmax):

def radial_gradient_magnitude(vmax,r,R):

def mean_shear_gradient(delta_v,separation):
    delta_v,=exact(delta_v);separation,=positive(separation);return abs(delta_v)/separation
def dynamic_viscosity(shear,rate):
    shear,rate=exact(shear,rate)
    if rate==0:raise ValueError('Viscosity is not identifiable from zero shear rate alone')
    eta=shear/rate
    if eta<=0:raise ValueError('Positive Newtonian viscosity branch required')
    return eta
def shear_stress(eta,rate):

def passive_hydraulic_power(drop,flow):
    drop,flow=exact(drop,flow)
    if drop*flow<0:raise ValueError('Passive pressure-work branch requires nonnegative power')
    return drop*flow
def series_resistance(*resistances):
    if not 1<=len(resistances)<=100:raise ValueError('Bounded nonempty resistance network required')
    return sum(positive(*resistances),sp.Integer(0))
def parallel_resistance(*resistances):
    if not 1<=len(resistances)<=100:raise ValueError('Bounded nonempty resistance network required')
    return 1/sum((1/v for v in positive(*resistances)),sp.Integer(0))
def unknown_parallel_resistance(equivalent,known):
    equivalent,known=positive(equivalent,known)
    if equivalent>=known:raise ValueError('Positive finite second branch requires equivalent<known')
    return 1/(1/equivalent-1/known)
def conductance(resistance):

def narrow_flow_fraction(wide_radius_factor):

    return 1/(1+wide_radius_factor**4)
def surface_work(gamma,new_total_area):

def film_force(gamma,border_length,interfaces=2):
    gamma,border_length=positive(gamma,border_length)
    if type(interfaces)is not int or interfaces not in(1,2):raise ValueError('One interface or two film faces required')
    return interfaces*gamma*border_length
def capillary_height(gamma,rho,g,radius,contact_cosine):
    gamma,rho,g,radius=positive(gamma,rho,g,radius);contact_cosine,=exact(contact_cosine)
    if not -1<=contact_cosine<=1:raise ValueError('Contact-angle cosine must be in[-1,1]')
    return 2*gamma*contact_cosine/(rho*g*radius)
def laplace_spherical_pressure(gamma,radius,interfaces=1):
    gamma,radius=positive(gamma,radius)
    if type(interfaces)is not int or interfaces not in(1,2):raise ValueError('One spherical interface or thin two-face film required')
    return 2*interfaces*gamma/radius
