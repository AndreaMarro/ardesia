from pathlib import Path
import subprocess,time,json,hashlib,sys
D=Path('/private/tmp/cric-author-a-resident/officina-di-fisica/editorial/reviews/c6-remaining-four-abb2-independent-a-20261127')
for name,cmd in [('static-supervised',[sys.executable,'/private/tmp/run_c6_abb2_static_a.py']),('analysis',[sys.executable,'/private/tmp/analyze_c6_abb2_a.py'])]:
